cmake_minimum_required(VERSION 3.16)
project(latest_camera_jpeg LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(JPEG REQUIRED)
add_executable(latest_camera_jpeg latest_camera_jpeg.cpp)
target_compile_features(latest_camera_jpeg PRIVATE cxx_std_17)
ament_target_dependencies(latest_camera_jpeg rclcpp sensor_msgs)
target_link_libraries(latest_camera_jpeg JPEG::JPEG)
